async function aprovarSolicitacao(requestId) {
  const reqRef = doc(db, 'requests', requestId);
  const reqSnap = await getDoc(reqRef);
  const dados = reqSnap.data();

  // modelo simples: sem Auth, só Firestore
  const userRef = doc(db, 'users', requestId); // ou outro ID
  await setDoc(userRef, {
    nome: dados.nome,
    email: dados.email,
    role: 'member',
    createdAt: new Date()
  });

  await updateDoc(reqRef, {
    status: 'approved',
    approvedAt: new Date()
  });

  // atualizar UI depois
}
